145def setup_compat_shaders_dir(path: str) -> dict:
146 """Setup per-game shader cache if shader pre-caching is disabled"""
147 environ = {}
148 shader_cache_name = 'steamapp_shader_cache'
149 shader_cache_vars = {
150 # Nvidia
151 '__GL_SHADER_DISK_CACHE_APP_NAME': shader_cache_name,
152 '__GL_SHADER_DISK_CACHE_PATH': os.path.join(path, 'nvidiav1'),
153 '__GL_SHADER_DISK_CACHE_READ_ONLY_APP_NAME': 'steam_shader_cache;steamapp_merged_shader_cache',
154 '__GL_SHADER_DISK_CACHE_SIZE': '10737418240', # 10GiB
155 # "__GL_SHADER_DISK_CACHE_SKIP_CLEANUP": "1",
156 # Mesa
157 'MESA_DISK_CACHE_READ_ONLY_FOZ_DBS': 'steam_cache,steam_precompiled',
158 'MESA_DISK_CACHE_SINGLE_FILE': '1',
159 'MESA_GLSL_CACHE_DIR': path,
160 'MESA_GLSL_CACHE_MAX_SIZE': '10G',
161 'MESA_SHADER_CACHE_DIR': path,
162 'MESA_SHADER_CACHE_MAX_SIZE': '10G',
163 # AMD VK
164 'AMD_VK_PIPELINE_CACHE_FILENAME': shader_cache_name,
165 'AMD_VK_PIPELINE_CACHE_PATH': os.path.join(path, 'AMDv1'),
166 'AMD_VK_USE_PIPELINE_CACHE': '1',
167 # DXVK
168 'DXVK_STATE_CACHE_PATH': os.path.join(path, 'DXVK_state_cache'),
169 # VKD3D
170 'VKD3D_SHADER_CACHE_PATH': os.path.join(path, 'VKD3D_shader_cache'),
171 }
172 for key, value in shader_cache_vars.items():
173 if key in {
174 '__GL_SHADER_DISK_CACHE_PATH',
175 'MESA_GLSL_CACHE_DIR',
176 'MESA_SHADER_CACHE_DIR',
177 'AMD_VK_PIPELINE_CACHE_PATH',
178 'DXVK_STATE_CACHE_PATH',
179 'VKD3D_SHADER_CACHE_PATH',
180 }:
181 os.makedirs(value, exist_ok=True)
182 environ[key] = value
183 return environ

249def get_rare_executable(*, external: bool = False) -> list[str]:
250 """
251 Returns the command list to invoke Rare for different platforms and packaging solutions
252 When used with container based packaging, such as Flatpak or Snap, returns the command
253 invoke Rare outside of the container
254
255 :param external: if True return the command to invoke Rare through Flatpak or Snap, defaults to false
256 :return: command list
257 """
258 logger.debug(f'Trying to find executable: {sys.executable}, {sys.argv}')
259
260 if os.environ.get('SNAP') and external:
261 return ['snap', 'run', 'rare']
262 if os.environ.get('container') == 'flatpak' and external: # noqa: SIM112
263 return ['flatpak', 'run', 'io.github.dummerle.rare']
264
265 # lk: detect if nuitka
266 if '__compiled__' in globals():
267 executable = [sys.executable]
268 elif sys.argv[0].endswith('__main__.py'):
269 executable = [sys.executable, '-m', 'rare']
270 elif platform.system() in {'Linux', 'FreeBSD', 'Darwin'}:
271 if p := os.environ.get('APPIMAGE'):
272 executable = [p]
273 else:
274 if sys.executable == os.path.abspath(sys.argv[0]):
275 executable = [sys.executable]
276 else:
277 executable = [os.path.abspath(sys.argv[0])]
278 elif platform.system() == 'Windows':
279 executable = [sys.executable]
280
281 if sys.executable != os.path.abspath(sys.argv[0]):
282 executable.append(os.path.abspath(sys.argv[0]))
283
284 if executable[0].endswith('python.exe'):
285 # be sure to start console-less then
286 executable[0] = executable[0].replace('python.exe', 'pythonw.exe')
287
288 if executable[0].endswith('pythonw.exe'):
289 if executable[1].endswith('rare'):
290 executable[1] = executable[1] + '.exe'
291 else:
292 executable = [sys.executable]
293
294 executable[0] = os.path.abspath(executable[0])
295 return executable

298def create_desktop_link(app_name: str, app_title: str = '', link_name: str = '', link_type='desktop') -> bool:
299 """
300 Creates a desktop or start menu shortcut link
301
302 :param app_name:
303 app_name or "rare_shortcut" for Rare itself
304 :param app_title:
305 the title shown in the shortcut
306 (overrides to "Rare" for "rare_shortcut")
307 :param link_name:
308 the sanitized filename of the shortcut (use the folder_name attribute of RareGame)
309 (overrides to "Rare" for "rare_shortcut")
310 :param link_type:
311 "desktop" or "start_menu"
312 :return bool:
313 True if successful else False
314 """
315 # macOS is based on Darwin
316 if not desktop_links_supported():
317 logger.error(f'Shortcut creation is not available on {platform.system()}')
318 return False
319
320 if link_type not in ['desktop', 'start_menu']:
321 logger.error(f'Invalid link type {link_type}')
322 return False
323
324 for_rare = app_name == 'rare_shortcut'
325
326 if for_rare:
327 icon_path = resources_path.joinpath('images', f'Rare.{desktop_icon_suffix()}')
328 app_title = 'Rare'
329 link_name = 'Rare'
330 else:
331 icon_path = desktop_icon_path(app_name)
332 if not app_title or not link_name:
333 logger.error('Missing app_title or link_name')
334 return False
335
336 shortcut_path = desktop_link_path(link_name, link_type)
337 if not shortcut_path.parent.exists():
338 logger.error(f'Parent directory {shortcut_path.parent} does not exist')
339 return False
340 else:
341 logger.info(f'Creating shortcut for {app_title} at {shortcut_path}')
342
343 if platform.system() in {'Linux', 'FreeBSD'}:
344 executable = get_rare_executable(external=True)
345 executable = shlex.join(executable)
346 if not for_rare:
347 executable = f'{executable} launch {app_name}'
348
349 with shortcut_path.open(mode='w') as desktop_file:
350 desktop_file.write(
351 '[Desktop Entry]\n'
352 f'Name={app_title}\n'
353 f'Comment={app_title} (Rare)\n'
354 'Type=Application\n'
355 'Categories=Game;\n'
356 f'Icon={icon_path}\n'
357 f'Exec={executable}\n'
358 'Terminal=false\n'
359 'StartupWMClass=Rare\n'
360 )
361 # shortcut_path.chmod(0o755)
362 return True
363
364 elif platform.system() == 'Windows':
365 # Add shortcut
366 shell = Dispatch('WScript.Shell')
367 shortcut = shell.CreateShortCut(str(shortcut_path))
368
369 executable = get_rare_executable()
370 arguments = []
371
372 if len(executable) > 1:
373 arguments.extend(executable[1:])
374 executable = executable[0]
375
376 if not for_rare:
377 arguments.extend(['launch', app_name])
378 shortcut.Targetpath = executable
379 # Maybe there is a better solution, but windows does not accept single quotes (Windows is weird)
380 shortcut.Arguments = shlex.join(arguments).replace("'", '"')
381
382 shortcut.Description = app_title
383 if for_rare:
384 shortcut.WorkingDirectory = str(home_dir())
385
386 # Icon
387 shortcut.IconLocation = str(icon_path.absolute())
388
389 shortcut.save()
390 return True
391
392 return False
