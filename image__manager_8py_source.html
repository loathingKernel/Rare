316 def _convert_image(self, image_data, logo_data, preset: ImageSize.Preset) -> QImage:
317 image = QImage()
318 image.loadFromData(image_data)
319 image.convertToFormat(QImage.Format.Format_ARGB32_Premultiplied)
320 if image.height() > image.width() and preset.orientation == ImageType.Wide:
321 image = image.transformed(QTransform().rotate(90.0))
322 if image.height() < image.width() and preset.orientation == ImageType.Tall:
323 image = image.transformed(QTransform().rotate(-90.0))
324 # lk: Images are not always at the correct aspect ratio, so crop them to size
325 wr, hr = preset.aspect_ratio
326 factor = min(image.width() // wr, image.height() // hr)
327 rem_w = (image.width() - factor * wr) // 2
328 rem_h = (image.height() - factor * hr) // 2
329 image = image.copy(rem_w, rem_h, factor * wr, factor * hr)
330
331 if logo_data is not None:
332 logo = QImage()
333 logo.loadFromData(logo_data)
334 logo.convertToFormat(QImage.Format.Format_ARGB32_Premultiplied)
335 if logo.width() > image.width():
336 logo = logo.scaled(
337 image.width(),
338 image.height(),
339 Qt.AspectRatioMode.KeepAspectRatio,
340 Qt.TransformationMode.SmoothTransformation,
341 )
342 painter = QPainter(image)
343 painter.drawImage(
344 (image.width() - logo.width()) // 2,
345 image.height() - logo.height(),
346 logo,
347 )
348 painter.end()
349
350 return image.scaled(
351 preset.size,
352 Qt.AspectRatioMode.KeepAspectRatio,
353 Qt.TransformationMode.SmoothTransformation,
354 )

356 def _convert_icon(self, cover: QImage) -> QImage:
357 icon_size = QSize(128, 128)
358 icon = QImage(icon_size, QImage.Format.Format_ARGB32_Premultiplied)
359 painter = QPainter(icon)
360 painter.setRenderHint(QPainter.RenderHint.SmoothPixmapTransform, True)
361 painter.setRenderHint(QPainter.RenderHint.Antialiasing, True)
362 painter.setCompositionMode(QPainter.CompositionMode.CompositionMode_Source)
363 painter.fillRect(icon.rect(), Qt.GlobalColor.transparent)
364 overlay = self._generate_icon_overlay_generate_icon_overlay(icon.rect())
365 brush = QBrush(cover)
366 scale = max(icon.width() / cover.width(), icon.height() / cover.height())
367 transform = QTransform().scale(scale, scale)
368 brush.setTransform(transform)
369 painter.fillPath(overlay, brush)
370 pen = QPen(Qt.GlobalColor.black, 2)
371 painter.setPen(pen)
372 painter.drawPath(overlay)
373 painter.end()
374 return icon
