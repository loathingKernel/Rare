63def load_color_scheme(path: str) -> QPalette:
64 palette = QPalette()
65 scheme = QSettings(path, QSettings.Format.IniFormat)
66 try:
67 scheme.beginGroup('ColorScheme')
68 for g in color_group_map:
69 scheme.beginGroup(color_group_map[g])
70 group = QPalette.ColorGroup(g)
71 for r in color_role_map:
72 role = QPalette.ColorRole(r)
73 color = scheme.value(color_role_map[r], None)
74 if color is not None:
75 palette.setColor(group, role, QColor(color))
76 else:
77 palette.setColor(group, role, palette.color(QPalette.ColorGroup.Active, role))
78 scheme.endGroup()
79 scheme.endGroup()
80 except Exception:
81 palette = None
82 return palette

179def relative_date(date):
180 diff = datetime.now(UTC) - date
181 s = diff.seconds
182 if diff.days > 7 or diff.days < 0:
183 return date.strftime('%d %b %y')
184 elif diff.days == 1:
185 return '1 day ago'
186 elif diff.days > 1:
187 return f'{diff.days} days ago'
188 elif s <= 1:
189 return 'just now'
190 elif s < 60:
191 return f'{s} seconds ago'
192 elif s < 120:
193 return '1 minute ago'
194 elif s < 3600:
195 return f'{s // 60} minutes ago'
196 elif s < 7200:
197 return '1 hour ago'
198 else:
199 return f'{s // 3600} hours ago'
