102 def _override_exe_edit_callback(self, path: str) -> tuple[bool, str, int]:
103 if not path or self.igameigame is None:
104 return True, path, IndicatorReasonsCommon.VALID
105 if not os.path.isabs(path):
106 path = os.path.join(self.igameigame.install_path, path)
107 # lk: Compare paths through python's commonpath because in windows we
108 # cannot compare as strings
109 # antonia disapproves of this
110 if os.path.commonpath([self.igameigame.install_path, path]) != self.igameigame.install_path:
111 return False, self.igameigame.install_path, IndicatorReasonsCommon.WRONG_PATH
112 if not os.path.exists(path):
113 return False, path, IndicatorReasonsCommon.WRONG_PATH
114
115 if not path.endswith(tuple(map(lambda s: s.replace('*', ''), self.override_exe_name_filters))):
116 return False, path, IndicatorReasonsCommon.WRONG_PATH
117 path = os.path.relpath(path, self.igameigame.install_path)
118 return True, path, IndicatorReasonsCommon.VALID
