19def read_registry(registry: str, prefix: str) -> ConfigParser:
20 accepted = ['system.reg', 'user.reg']

29def prepare_process(command: list[str], environment: dict) -> tuple[str, list[str], dict]:
30 logger.debug('Preparing process: %s', command)
31 _env = os.environ.copy()
32 _command = command.copy()
33 if os.environ.get('container') == 'flatpak': # noqa: SIM112
34 flatpak_command = ['flatpak-spawn', '--host']
35 flatpak_command.extend(f'--env={name}={value}' for name, value in environment.items())
36 _command = flatpak_command + command
37 else:
38 _env.update(environment)
39 return _command[0], _command[1:] if len(_command) > 1 else [], _env

49def get_configured_qprocess(command: list[str], environment: dict) -> QProcess:

72def execute_subprocess(command: list[str], arguments: list[str], environment: Mapping) -> tuple[str, str]:
